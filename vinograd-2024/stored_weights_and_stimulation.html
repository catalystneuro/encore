The claims file was written from the version of record (open access, corrected 2025) with its Methods and the legends of the Extended Data figures.

Dataset: DANDI:001037, version 0.260609.1402. The paper cites the dandiset without a version. Of the three published versions, 0.240816.1841 holds 2 files, 0.260529.1406 holds 9 (no miniscope mice except M30L) and 0.260609.1402 holds 13: the five miniscope mice and the seven two-photon mice. It is the most recent and the only one that holds the data of every claim the dandiset can test, so it is pinned. The data check below lists the files of each version.

def distances(d, ses, tinit, which):
    s = np.c_[L.projection(d, ses, "x1"), L.projection(d, ses, "x2")]
    on = d["stim"][ses]; isi = on[1] - on[0]
    p0 = s[:on[0]].mean(0) if tinit == "mean" else s[on[0] - 1]
    out = [np.linalg.norm(s[:on[0]] - p0, axis=1).mean()]
    for i, o in enumerate(on):
        tt = o + L.PULSE_FRAMES if which == "stim" else (on[i + 1] - 1 if i + 1 < len(on) else o + isi)
        out.append(np.linalg.norm(s[min(tt, len(s) - 1)] - p0))
    return np.array(out)

from scipy.optimize import minimize_scalar

def step_fit(tr, on, cap):
    start = on[0]
    y = tr[start:]
    k = np.zeros(len(y))
    for o in on:
        k[o - start:] += 1
    def mse_for(c):
        lev = k if c is None else np.minimum(k, c)
        s = (lev @ y) / (lev @ lev)
        return np.mean((y - s * lev) ** 2), s * lev
    if not cap:
        e, pred = mse_for(None)
    else:
        r = minimize_scalar(lambda c: mse_for(c)[0], bounds=(0.5, 4.0), method="bounded")
        e, pred = mse_for(r.x)
    after = on[2] + L.PULSE_FRAMES - start
    return e, np.mean((y[after:] - pred[after:]) ** 2)
