Data check¶

What the files hold, checked against the paper's description (Methods, "Neuropixels recording and processing"; claims file, opening comment). The clock check compares the mean z-scored population response 0 to 200 ms after each stimulus onset of the table (minus the 1 s before) with the same measure at onsets shifted by 1.5 s, between stimuli: a response at the table's onsets and none at the shifted ones shows that spike times and stimulus table share a clock. The second cell (added in revision) places the puff series against the file's infusion time and counts tone and light onsets inside the windows the analyses use.

def peak_decay(trace, frac=0.25):
    """analysis.py find_decay: |trace| normalized by its largest |value| after onset; the peak is its argmax and
    the decay the time from onset until it falls below frac after the peak; a peak whose z-score (over the
    trace) is below 1 in absolute value gets decay 2 s, as the code does. Returns peak value (signed),
    peak time, decay."""
    zt = scipy.stats.zscore(trace)
    a = np.abs(trace / np.nanmax(np.abs(trace[100:])))
    pk = int(np.argmax(a[100:]))
    below = np.where(a[100:][pk:] < frac)[0]
    decay = (pk + below[0]) * K.BIN if len(below) else np.nan
    if abs(zt[100:][pk]) < 1:
        decay = 2.0
    return trace[100 + pk], pk * K.BIN, decay

def overrep(labels, nclus, seed=2024):
    """analysis.py get_cluster_null_distribution: per region with more than 30 clustered cells, the fraction
    in each cluster against 20,000 uniformly random labelings (label 0 of the draws), two-sided; BH-FDR
    across regions. Returns the corrected p (regions x clusters), the fractions and the regions."""
    rng = np.random.default_rng(seed)
    valid = mod_regions != "NA"
    ureg, ucnt = np.unique(mod_regions[valid], return_counts=True)
    big = ureg[ucnt > 30]
    pvals, fracs = {}, {}
    for r in big:
        ncell = (mod_regions == r).sum()
        null = (rng.integers(nclus, size=(20000, ncell)) == 0).mean(1)
        obs = np.bincount(labels[mod_regions == r], minlength=nclus) / ncell
        pvals[r] = np.array([np.mean(np.abs(null - 1 / nclus) >= abs(o - 1 / nclus)) for o in obs])
        fracs[r] = obs
    P = pd.DataFrame(pvals).T
    F = pd.DataFrame(fracs).T
    Q = P.apply(lambda col: pd.Series(scipy.stats.false_discovery_control(np.maximum(col.values, 1 / 20000)), index=col.index))
    return Q, F, big

def figs31b_test(Q, F, big, nclus, dec, thr):
    """PAG and VPM in a faster-half cluster and a frontal/limbic region in a slower-half cluster, with clusters
    ranked by `dec` (decay to 25% of the peak). With an odd count the faster half is the nclus // 2 fastest."""
    fast_half = set(np.argsort(np.nan_to_num(dec, nan=99), kind="stable")[: nclus // 2])
    over = (Q < thr) & (F > 1 / nclus)
    pag = [int(c) for c in np.where(over.loc["PAG"])[0] if c in fast_half] if "PAG" in big else []

    return over, fast_half, pag, vpm, fr

def dprime_abs(p):
    """region_time_course calculate_dprime on (..., trials, 300): |mean - baseline mean| / pooled sd."""
    bm = p[..., :100].mean((-1, -2))
    bs = np.nanstd(p[..., :100].reshape(p.shape[:-2] + (-1,)), -1)
    rm = p.mean(-2)
    rs = np.nanstd(p, -2)
    return np.abs(rm - bm[..., None]) / np.sqrt(0.5 * (bs[..., None] ** 2 + rs ** 2))

def rise_decay(p, thr=0.25):
    """region_time_course _find_rise_decay, vectorized over leading axes of p (..., trials, 300)."""
    d = dprime_abs(p)
    d = d - d[..., :100].mean(-1, keepdims=True)
    d = d / d.max(-1, keepdims=True)
    post = d[..., 100:]
    pk = post.argmax(-1)
    above = post > thr
    rise = np.where(above.any(-1), above.argmax(-1) * K.BIN, np.nan)
    idx = np.arange(post.shape[-1])
    after = (post < thr) & (idx >= pk[..., None])
    decay = np.where(after.any(-1), after.argmax(-1) * K.BIN, np.nan)
    bad = pk > 75
    rise[bad] = np.nan
    decay[bad] = np.nan
    return rise, decay

def degenerate(b):
    """A fit whose timescale is not finite, not positive, or beyond the 2.5 s fit window."""
    b = np.asarray(b, float)
    return ~np.isfinite(b) | (b <= 0) | (b > 2.5)

def timescale_rows(get_sigs, ks, exp_pre, exp_ket, labels):
    """Per-series fits averaged (mean, as the Methods, and median) and one concatenated fit, under each fit
    variant; paired t-test across sessions. get_sigs(k) returns (Pre signals, Ketamine signals)."""
    rows = []
    for fv, kw in FIT_VARIANTS.items():
        a_pre, a_ket, m_pre, m_ket, b_pre, b_ket, ndeg, nfit = [], [], [], [], [], [], 0, 0
        for k in ks:
            ps, kt = get_sigs(k)
            tp = np.array([K.extract_timescale(s, **kw) for s in ps]); tk = np.array([K.extract_timescale(s, **kw) for s in kt])
            ndeg += degenerate(tp).sum() + degenerate(tk).sum(); nfit += len(tp) + len(tk)
            a_pre.append(np.nanmean(tp)); a_ket.append(np.nanmean(tk))
            m_pre.append(np.nanmedian(tp)); m_ket.append(np.nanmedian(tk))
            b_pre.append(K.extract_timescale(ps.reshape(-1), **kw)); b_ket.append(K.extract_timescale(kt.reshape(-1), **kw))
        for lab, p_, k_ in (("(a) per series, mean", a_pre, a_ket), ("(a') per series, median", m_pre, m_ket), ("(b) concatenated", b_pre, b_ket)):
            p_, k_ = np.array(p_), np.array(k_)
            t = scipy.stats.ttest_rel(p_, k_)
            rows.append(dict(labels, fit=fv, combine=lab, Pre=p_.mean(), Ket=k_.mean(), t=t.statistic, p=t.pvalue,
                             holds=bool(t.pvalue < 0.05 and k_.mean() < p_.mean()),
                             within=bool(abs(p_.mean() - exp_pre[0]) <= exp_pre[1] and abs(k_.mean() - exp_ket[0]) <= exp_ket[1]),
                             **{"degenerate per-series fits": f"{ndeg} of {nfit}" if lab.startswith("(a") else
                                f"{int(degenerate(p_).sum() + degenerate(k_).sum())} of {2 * len(p_)}"},
                             **{"Ket range": f"{k_.min():.3f}-{k_.max():.3f}"}))
    return rows

def emo_signal(o, cond, scale):
    """Projection over the series (8 puff windows, -1 to 23 s after the first puff) and the post-series window

    f = o["l1_factor"] if scale == "l1" else 1.0
    s8 = o[cond]["s8"] * f

def tone_vals(ks, split, key, puffs, win, base="global"):
    out = {}
    for c in ("pre", "ket", "post"):
        a = np.array([scipy.stats.zscore(res[k]["tone"][split]["proj"][key + (c,)][puffs]) for k in ks])
        a = a - (a[:, :100].mean() if base == "global" else a[:, :100].mean(1, keepdims=True))
        out[c] = a[:, win].mean(1)
    return out

def tone_tests(v):
    t1 = scipy.stats.ttest_rel(v["pre"], v["ket"]); t2 = scipy.stats.ttest_rel(v["post"], v["ket"])
    q = bh([t1.pvalue, t2.pvalue])
    return t1.statistic, q[0], t2.statistic, q[1]

def corr_table(pre_key, ket_key):
    rows = []
    for k in names:
        reg = res[k]["regions_all"]
        ok = reg != "NA"
        d = (res[k][ket_key] - res[k][pre_key])[np.ix_(ok, ok)].astype(float)
        r = reg[ok]
        same = r[:, None] == r[None, :]
        iu = np.triu(np.ones_like(same), 1).astype(bool)
        fin = np.isfinite(d)
        row = {"session": k, "units": int(ok.sum()), "within change": np.nanmean(d[same & iu & fin]), "between change": np.nanmean(d[~same & iu & fin]),
               "pairs dropped (silent unit)": int((iu & ~fin).sum())}
        for fam in ("RSP", "ORB"):
            m = np.array([K.under(x, fam, paths) for x in r])
            row[f"{fam} units"] = int(m.sum())
            row[f"{fam} within change"] = np.nanmean(d[np.outer(m, m) & same & iu & fin]) if m.sum() > 1 else np.nan
            row[f"{fam} family change"] = np.nanmean(d[np.outer(m, m) & iu & fin]) if m.sum() > 1 else np.nan
        rows.append(row)
    return pd.DataFrame(rows).set_index("session")

How the analysis differs from the paper's:

  • The excluded animal of Figures 5 and 6 is not named; every reading (without sub-1, sub-3 or sub-5) is run, and where they disagree sub-5 is assumed, the only reading that also gives the 7 tone sessions from 7 mice of table S3.
  • The clustering gives 11 clusters, not 12, so the code's skipping of cluster 11 in the regional timing does not apply. Leiden's partition changes with its seed and, through the rounding of the float32 PCA, with the number of BLAS threads; BLAS is limited to one thread so that the partition does not depend on the machine, the outcomes are judged on that partition with scanpy's default Leiden seed, and ten seeds and two rounding-level perturbations are shown.
  • Rates are binned from 0 to the last spike of the session, where brainwide-npix uses its own recording length; z-scoring is over that span.
  • The tone epochs are split at the infusion time of the file (criterion) and at the code's ketamine onset, which falls before the infusion, so the code's tone Pre has 9 series, not 10.
  • The driver of the paper's code is not released; the post-series window (from 1 s before the last puff) and other alignments follow the claims file's reading of the plotting code.

What was found about the dataset:

  • Spike times are stored in no particular order within every one of the 9554 units; sorted, they are consistent with the stimulus table (the clock check shows a response at the table's onsets in every session and none at onsets shifted by 1.5 s).
  • The file has tone series in 9 sessions from 8 mice, where table S3 reports 7 from 7, and light stimuli (240 to 248 per session in 9 sessions) that the paper does not describe.
  • The infusion time in the files ("approximate") falls 94 to 420 s after the code's ketamine onset (30 s after the last puff of series 20).
  • 867 units have the region label NA; sub-5_ses-20230706 has two probes and 154 units.
  • No eye closure is in the dandiset; it is in DANDI:001347.
  • In the 9 sessions with tones and lights, light or tone onsets fall in the 30 s before 14 of the 30 puff series (Pre 1 to 20 and Ketamine 23 to 32; tone before series 5, 7, 9 and 29, light before 4, 6, 8, 14, 16, 20, 24, 28, 30 and 32) that the timescale and coupling analyses use, and in two sessions light onsets fall in every odd post-series window.
