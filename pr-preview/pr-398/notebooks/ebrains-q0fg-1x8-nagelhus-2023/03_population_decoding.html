Nagelhus et al. (2023), population decoding of the object¶

Paper. Nagelhus et al. (2023), Neuron 111, 2091-2104, doi:10.1016/j.neuron.2023.04.008; claims file written from the version of record with its STAR Methods and Supplemental information.

Dataset. EBRAINS 10.25493/Q0FG-1X8, version v1, the version whose DOI the paper cites and the only one published (data check of every file in 01_single_cells.ipynb).

The decoding sessions are the 16 sessions of the four population files: 14 tetrode sessions (rats 25987 and 26186, 25 to 39 cells) and the two Neuropixels sessions (135 and 620 cells). This is the legends' count (16 sessions, 14 tetrode). The Methods' text says 15 tetrode sessions, which the dataset does not hold, so the claims file's session readings (S-methods, S-legends) give the same 16 sessions. Every decoder runs under the claims file's two speed readings (speed a: rate maps and occupancy from epochs above 2.5 cm/s; b: no speed filter). Rate maps are built as the claims file now gives (answered 2026-10-21, from calculateRateMap.m): the rate per bin, unvisited bins 0, then smoothed and not blanked; tables print this as rm b.

PARTS = {}
for s in POP:
    for sp in "ab":
        PARTS[s.index, s.rat, sp] = (oc.half_parts(s.trials[0], sp == "a"), oc.half_parts(s.trials[1], sp == "a"))
CACHE = {}


def run(mode="distance", sp="a", rm="b", units=None, sessions=None, **kw):
    """Decoder on every decoding session; returns (sessions, bins) accuracies and the results."""
    out, res = [], []
    for s in (POP if sessions is None else sessions):
        pA, pB = PARTS[s.index, s.rat, sp]
        u = units(s) if callable(units) else units
        r = oc.decode_pair(pA, pB, mode, obj=s.trials[1].obj, rm=rm, units=u, cache=CACHE.setdefault((s.index, s.rat, sp), {}), **kw)
        out.append(r["acc"])
        res.append(r)
    return np.array(out), res


def kw_table(A):
    h, p = oc.kruskal(A.T)
    return h, p


MAIN = {}
for sp, rm in READINGS:
    MAIN[sp, rm] = run("distance", sp, rm, return_samples=True)

for (sp, rm), bl, reading in [(x, b, r) for x in READINGS for b in "ab" for r in "ab"]:
    vals, n_blank = [], []
    for s in POP:
        NO, O = s.trials
        mA = oc.rate_maps(*oc.raw_maps(NO, oc.sample_mask(NO, sp == "a"), 3.0), rm=rm, blank=bl == "a")
        mB = oc.rate_maps(*oc.raw_maps(O, oc.sample_mask(O, sp == "a"), 3.0), rm=rm, blank=bl == "a")
        r = oc.pv_correlation(mA, mB)
        n_blank.append(int(np.isnan(r).sum()))
        d = np.hypot(X3 - O.obj[0], Y3 - O.obj[1])
        if reading == "a":
            g = np.searchsorted([20, 40, 60, 80, 100], d, side="right")
            g[d >= 100] = -1
        else:
            order = np.argsort(d)
            g = np.empty(len(d), int)

def match_keep(groupsA, groupsB, rng):
    """Indices kept in A and B so that their counts match in every group."""
    keepA, keepB = [], []
    for g in np.union1d(groupsA[groupsA >= 0], groupsB[groupsB >= 0]):
        ia, ib = np.flatnonzero(groupsA == g), np.flatnonzero(groupsB == g)
        n = min(len(ia), len(ib))
        keepA.append(rng.choice(ia, n, replace=False))
        keepB.append(rng.choice(ib, n, replace=False))
    return (np.concatenate(keepA) if keepA else np.zeros(0, int)), (np.concatenate(keepB) if keepB else np.zeros(0, int))


def behaviour_group(kind, speed, x, y):
    if kind == "speed":
        g = np.floor(speed / 5).astype(int)

    return np.clip(np.floor(x / 30), 0, 4).astype(int) * 5 + np.clip(np.floor(y / 30), 0, 4).astype(int)
