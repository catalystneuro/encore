Reproducing Hazon et al. (2022) from CRCNS hc-30: position decoding¶

Paper. Hazon O, Minces VH, Tomàs DP, Ganguli S, Schnitzer MJ, Jercog PE (2022). Noise correlations in neural ensemble activity limit the accuracy of hippocampal spatial representations. Nature Communications 13, 4276. doi:10.1038/s41467-022-31254-y

Dataset. CRCNS hc-30, Jercog P and Schnitzer M (2022), "Functional calcium imaging recordings of hippocampal area CA1 from mice performing a linear track task", doi:10.6080/K0GH9G5X. One file, hc-30/live/event_traces.mat (142,350,100 bytes), streamed from the public S3 bucket crcnsarchive into memory.

Version. CRCNS data sets are not versioned, so the version is recorded as unversioned (docs/archives.md). The paper cites no version; the only one published is the one in the bucket (data descriptor version 0.6, July 8, 2022, the initial version), which matches the claims file's description (110 sessions with mouse name, session ID, sparse event array and position). The MD5 sum from the data set's checksums.md5, which the next cell checks against the streamed bytes, is fe0f5dd2fca6d64dbf0cd3f35e31e1f0 (event_traces.mat); the descriptor PDF is 320d157fbcc43be385485cc186842636.

Text the claims were written from. The version of record (PMC9314334) with its Methods and Supplementary Information; the paper's code, github.com/approbatory/ML-project at commit 2057af9facfd3035c573e1254bdc32ed4781ba60, was read for method details (claims/hazon-2022.yaml). Method details taken by this attempt cite the passage or code file in hc30.py.

Data. 110 sessions of 12 mice with an event-onset array (frames x neurons) and the position in camera pixels. There are no cell positions, no frame times and no trial table; trials, bins and the event trace are built here from the Methods and the code (hc30.py).

Decoding runs¶

Every session: REPS_FULL repetitions with all neurons, and REPS_SUB subsets at each ensemble size below the session's neuron count. Each session has its own seed.

def job(i):
    T, d = TENS[i]
    full = hc30.decode_full(T, d, seed=20000 + i, reps=REPS_FULL, C=C_SVM)
    sizes = [s for s in SIZES if s < T.shape[0]]
    iu, is_ = hc30.decode_series(T, d, seed=30000 + i, sizes=sizes, reps=REPS_SUB, C=C_SVM)
    full["sizes"] = np.array(sizes + [T.shape[0]])
    full["imse_u"] = np.r_[iu, np.mean(1 / np.array(full["mse_u"]))]
    full["imse_s"] = np.r_[is_, np.mean(1 / np.array(full["mse_s"]))]
    return full

t = time.time()
with mp.get_context("fork").Pool(4) as pool:

def per_bin_rms(conf):
    pos = np.arange(40) // 2
    err2 = (pos[:, None] - pos[None, :]) ** 2 * hc30.BIN_CM ** 2
    by_class = np.sqrt((conf * err2).sum(1) / conf.sum(1))
    by_pos = np.sqrt(np.array([(conf[2 * b:2 * b + 2] * err2[2 * b:2 * b + 2]).sum()
                               / conf[2 * b:2 * b + 2].sum() for b in range(20)]))
    return by_pos, by_class

rng9 = np.random.default_rng(909)
L9, FS9, SPEED = hc30.TRACK_CM, 20.0, 30.0
n_frames = int(30 * 60 * FS9)
step = SPEED / FS9
phase = (np.arange(n_frames) * step) % (2 * L9)
true_pos = np.where(phase < L9, phase, 2 * L9 - phase)
direction = np.where(phase < L9, 1, -1)
run_id = (np.arange(n_frames) * step // L9).astype(int)
noisy_pos = true_pos + rng9.normal(0, 10, n_frames)
true_bin = np.clip((true_pos / L9 * 20).astype(int), 0, 19)
cls = 2 * true_bin + (direction == -1)
sigma9 = 30 / (2 * np.sqrt(2 * np.log(2)))
key = run_id * 40 + cls
uk, inv = np.unique(key, return_inverse=True)
s_run, s_cls = uk // 40, uk % 40
s_frames = np.bincount(inv).astype(float)
n_runs = run_id.max() + 1
print(f"{n_runs} runs, {len(uk)} run-bin samples, {s_frames.mean():.1f} frames per sample")

def simulate(n_cells):
    c = rng9.uniform(0, L9, n_cells)
    Y = np.zeros((len(uk), n_cells))
    for a in range(0, n_frames, 6000):
        rate = 0.05 * np.exp(-0.5 * ((noisy_pos[a:a + 6000, None] - c[None, :]) / sigma9) ** 2)
        np.add.at(Y, inv[a:a + 6000], rng9.poisson(rate))
    return Y

def nb_imse(Y, sizes, reps):
    out = []
    for n in sizes:
        v = []
        for _ in range(reps):
            sub = rng9.choice(Y.shape[1], n, replace=False)
            first_runs = rng9.permutation(n_runs)[: n_runs // 2]
            half = np.isin(s_run, first_runs)
            mses = []
            for a, b in ((half, ~half), (~half, half)):
                Ya, Yb = Y[a][:, sub], Y[b][:, sub]
                lam = np.array([Ya[s_cls[a] == k].sum(0) / s_frames[a][s_cls[a] == k].sum()
                                for k in range(40)]) + 1e-6
                ll = Yb @ np.log(lam).T - s_frames[b][:, None] * lam.sum(1)[None, :]
                p = ll.argmax(1)
                mses.append(np.mean(((p // 2 - s_cls[b] // 2) * hc30.BIN_CM) ** 2))
            v.append(1 / np.mean(mses))
        out.append(np.mean(v))
    return np.array(out)
