Reproducing Park et al. (2019) from CRCNS hc-20¶

Paper. Park EH, Keeley S, Savin C, Ranck JB Jr, Fenton AA (2019). How the internally organized direction sense is used to navigate. Neuron 101:285-293. doi:10.1016/j.neuron.2018.11.019. The claims file (claims/park-2019.yaml) was written from the version of record with its STAR Methods and Supplemental Information, and from the paper's code (https://github.com/FentonLab/PyCode_MEC_HDC-NeuronMS_Park at commit 545a738).

Data set. CRCNS hc-20, doi:10.6080/K0SF2TDN, the file hc-20/live/data.zip in the public S3 bucket crcnsarchive, streamed and opened in place. CRCNS data sets are not versioned, and the paper cites the data set's DOI only, so the version pinned is the one published now, recorded as unversioned with the MD5 sum of data.zip from the data set's checksums.md5 (b82c7b97b0ba3491f04ecc4dbcd48e83); the setup cell below checks it against the published file, so a later change is detectable. There is no other version to choose between. The data descriptor is version 0.6 (14 Nov 2018).

What the data set holds. The spike trains and 30 Hz tracking of the 115 head-direction cells in the three trials (stable1, rotating in room and arena coordinates, stable2), a file list that matches units across trials, and the tracker's files of each trial. It holds no other MEC units, no LFP, no waveforms and no training sessions.

def zone_xy(z_):
    """Pixel position of the centre of a sector (centre angle, width, inner, outer radius), the
    angle counted with y up, as the flagged samples show."""
    a = np.deg2rad(z_[0])
    r = (z_[2] + z_[3]) / 2
    return 127.5 + r * np.cos(a), 127.5 - r * np.sin(a)

def speed(x, step):
    ok = ~((x["x"] == 0) & (x["y"] == 0))
    px, py = x["x"].astype(float), x["y"].astype(float)
    px[~ok] = np.nan
    py[~ok] = np.nan
    dist = np.hypot(px[step:] - px[:-step], py[step:] - py[:-step])
    return np.nanmean(dist) / 2.1333 * x["fs"] / step   # cm/s

def interval_xy(x, n, w=4):
    nw = n // w
    px = x["x"][: nw * w].astype(float).reshape(nw, w)
    py = x["y"][: nw * w].astype(float).reshape(nw, w)
    ok = ~((px == 0) & (py == 0))
    with np.errstate(invalid="ignore"):
        return np.where(ok, px, 0).sum(1) / ok.sum(1), np.where(ok, py, 0).sum(1) / ok.sum(1)


def zone_distance(x, y, z_, kind):
    """Distance (pixels, or degrees for 'angle') from positions to a sector zone (centre angle, width,
    inner, outer radius; angles counted with y up): to its centre, to its nearest point (0 inside),
    or the angle about the arena centre between the position and the zone's centre direction."""
    dx, dy = x - 127.5, -(y - 127.5)
    if kind == "centre":
        cx, cy = zone_xy(z_)
        return np.hypot(x - cx, y - cy)
    th = np.rad2deg(np.arctan2(dy, dx)) % 360
    if kind == "angle":
        return hd20.circ_absdiff(th, z_[0])
    r = np.hypot(dx, dy)
    inside = hd20.circ_absdiff(th, z_[0]) <= z_[1] / 2
    d_in = np.maximum.reduce([z_[2] - r, r - z_[3], np.zeros_like(r)])
    d_edge = []
    for phi in np.deg2rad([z_[0] - z_[1] / 2, z_[0] + z_[1] / 2]):
        t = np.clip(dx * np.cos(phi) + dy * np.sin(phi), z_[2], z_[3])   # nearest point on the edge segment
        d_edge.append(np.hypot(dx - t * np.cos(phi), dy - t * np.sin(phi)))
    return np.where(inside, d_in, np.minimum(*d_edge))

def shift_r(a, b):
    a, b = a - a.mean(), b - b.mean()
    c = np.fft.irfft(np.fft.rfft(a) * np.conj(np.fft.rfft(b)), len(a))

def circ_r(deg):
    deg = np.asarray(deg, float)
    deg = deg[~np.isnan(deg)]
    return float(abs(np.exp(1j * np.deg2rad(deg)).mean())), len(deg)


def error_distributions(err, hd):
    """The three distribution readings of the criterion, each as (r, n for Rayleigh, Kuiper V, Kuiper p)."""
    out = {}
    b = (hd // 10).astype(int) % 36
    occ = np.bincount(b, minlength=36)
    # (i) mean absolute error per 10 degree head-direction bin, as weights over the bin centres
    mean_err = np.bincount(b, weights=np.abs(err), minlength=36) / np.maximum(occ, 1)
    r, _ = hd20.resultant(mean_err)
    out["(i) error by head direction"] = (r, len(err), *hd20.kuiper_binned(mean_err, occ, len(err), len(err)),
                                          int((occ > 0).sum()))
    # (ii) head directions of windows decoded within 15, 30 or 45 degrees
    for thr in [15, 30, 45]:
        acc = hd[np.abs(err) <= thr]
        r, n = circ_r(acc)
        out[f"(ii) head direction, error <= {thr}"] = (r, n, *hd20.kuiper_two_sample(acc, hd), None)
    # (iii) the signed error angles
    r, n = circ_r(err)
    out["(iii) signed error"] = (r, n, *hd20.kuiper_two_sample(err, hd), None)
    return out

def two_minima(y):
    y = np.asarray(y, float)
    loc = [i for i in range(36) if y[i] < y[i - 1] and y[i] < y[(i + 1) % 36]]
    depth = {i: min(y[i - 1], y[(i + 1) % 36]) - y[i] for i in loc}
    return sorted(loc, key=lambda i: y[i])[:2], depth

How the analysis differs from the paper's. Spike width cannot be applied (no waveforms); the 115 deposited units are taken as the head-direction cells. Units are matched across trials by file name where FileNameList.xlsx is shifted. The decoder and the internally referenced tuning are ports of the paper's Python code; Idir, relative proximity, the strength of internal tuning and several test details are not fully specified and are run under the readings the claims file names. Statistics use Python in place of JMP 12, with the Greenhouse-Geisser correction applied when Mauchly's test is significant. After repeated-measures ANOVAs, Tukey HSD uses the ANOVA's repeated-measures error term (an assumption; the between-groups Tukey-Kramer HSD is shown as a variant); after one-way ANOVAs it is the between-groups Tukey-Kramer HSD.
